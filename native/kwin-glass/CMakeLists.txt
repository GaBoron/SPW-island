cmake_minimum_required(VERSION 3.22)
project(spw_island_kwin_glass LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus OpenGL)
find_package(VulkanHeaders CONFIG QUIET)
find_package(KWin CONFIG REQUIRED)
if(KWin_VERSION VERSION_LESS 6.6 OR KWin_VERSION VERSION_GREATER_EQUAL 6.9)
    message(FATAL_ERROR "SPW Island currently targets KWin 6.6 through 6.8")
endif()
if(NOT KWin_VERSION VERSION_EQUAL SPW_EXPECTED_KWIN_VERSION)
    message(FATAL_ERROR "The installed KWin SDK must match the running compositor")
endif()
if(NOT SPW_PLUGIN_NAME MATCHES "^spw_island_glass_[a-f0-9]+$")
    message(FATAL_ERROR "A unique SPW_PLUGIN_NAME is required")
endif()
if(NOT SPW_NATIVE_REVISION MATCHES "^[a-f0-9]+$")
    message(FATAL_ERROR "SPW_NATIVE_REVISION is required")
endif()
add_library(${SPW_PLUGIN_NAME} MODULE main.cpp IslandGlassEffect.cpp GlassRenderer.cpp)
target_link_libraries(${SPW_PLUGIN_NAME} PRIVATE KWin::kwin Qt6::DBus Qt6::OpenGL epoxy::epoxy)
target_compile_definitions(${SPW_PLUGIN_NAME} PRIVATE
    SPW_NATIVE_REVISION="${SPW_NATIVE_REVISION}" SPW_KWIN_VERSION="${KWin_VERSION}"
    SPW_PLUGIN_NAME="${SPW_PLUGIN_NAME}")
if(KWin_VERSION VERSION_GREATER_EQUAL 6.7)
    target_compile_definitions(${SPW_PLUGIN_NAME} PRIVATE SPW_KWIN_67)
endif()
if(KWin_VERSION VERSION_GREATER_EQUAL 6.8)
    target_compile_definitions(${SPW_PLUGIN_NAME} PRIVATE SPW_KWIN_68)
endif()
set_target_properties(${SPW_PLUGIN_NAME} PROPERTIES PREFIX "")
